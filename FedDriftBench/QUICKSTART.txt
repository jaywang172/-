🚀 INSTALLATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Install dependencies:
   pip install -r requirements.txt

2. Verify installation:
   python -c "import torch; import river; print('✓ Ready!')"


🎯 RUN EXPERIMENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

OPTION 1: Quick Demo (5 minutes)
   python quick_start.py

OPTION 2: Full Benchmark (1-2 hours)
   python experiments/run_benchmark.py

OPTION 3: Custom Experiment
   python examples/custom_experiment.py --mode detector

After running experiments:
   python experiments/visualize_results.py

Plots saved to: results/visualizations/

🔬 4 BENCHMARK SCENARIOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

A. Homogeneous Drift      - All clients drift simultaneously
B. Heterogeneous Drift    - Clients drift at different times
C. Partial Participation  - Subset of clients each round
D. Staggered Drift        - Groups drift in sequence


🛠️  3 DRIFT DETECTORS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• ADWIN  - Adaptive windowing (best for gradual drift)
• DDM    - Error monitoring (fast sudden drift detection)
• KSWIN  - Statistical testing (balanced performance)


📊 EVALUATION METRICS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• Precision          - Correctness of detections
• Recall             - Coverage of true drifts
• F1 Score           - Balanced metric
• Detection Delay    - Speed of detection
• Final Accuracy     - End-to-end performance
